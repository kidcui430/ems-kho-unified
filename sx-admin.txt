let routingStepCount = 0;

window.addRoutingStep = function () {
  routingStepCount++;
  const container = el("routing-builder");
  const stepDiv = document.createElement("div");
  stepDiv.style.display = "flex";
  stepDiv.style.gap = "10px";

  container.appendChild(stepDiv);
};

export async function fetchProductsList() {
  try {
    const data = await api("/api/products");
    if (data.products) {
      state.productsCache = data.products;
      if (el("product-suggestions")) {
        el("product-suggestions").innerHTML = state.productsCache
          .map(
            (p) =>
              `<option value="${p.ma_sp}">${p.ma_sp} - ${p.ten_sp}</option>`,
          )
          .join("");
      }
      renderMasterTable(state.productsCache);
      bindPoProductCodePreview();
    }
  } catch (e) {
    console.error("Lỗi:", e);
  }
}

// Hiển thị diễn giải Tên SP & Lộ trình ngay khi gõ Mã Hàng ở form Tạo LOT
function bindPoProductCodePreview() {
  const codeInput = el("product-code");
  if (!codeInput) return;

  let previewEl = el("po-product-preview");
  if (!previewEl) {
    previewEl = document.createElement("div");
    previewEl.id = "po-product-preview";
    previewEl.style.cssText =
      "margin-top:6px; padding:8px 10px; border-radius:6px; font-size:13px; background:#f0f9ff; border:1px solid #bae6fd; display:none;";
    codeInput.parentElement.appendChild(previewEl);

    codeInput.addEventListener("input", function () {
      const code = this.value.trim().toUpperCase();
      const prod = (state.productsCache || []).find(
        (p) => String(p.ma_sp).toUpperCase() === code,
      );
      if (!code) {
        previewEl.style.display = "none";
        return;
      }
      previewEl.style.display = "block";
      if (!prod) {
        previewEl.style.background = "#fef2f2";
        previewEl.style.borderColor = "#fecaca";
        previewEl.innerHTML = `<span style="color:#dc2626; font-weight:bold;">⚠️ Không tìm thấy mã SP "${escapeHtml(code)}" trong danh mục!</span>`;
        return;
      }
      let routeStr = "Chưa khai báo lộ trình SX";
      if (prod.routing_map) {
        try {
          routeStr = JSON.parse(prod.routing_map)
            .map((step) => stageName(step.stage_id))
            .join(" ➔ ");
        } catch (e) {}
      }
      previewEl.style.background = "#f0f9ff";
      previewEl.style.borderColor = "#bae6fd";
      previewEl.innerHTML =
        `<div>🏷️ <b>Mã SP:</b> <b style="color:var(--navy);">${escapeHtml(prod.ma_sp)}</b> — <b style="color:#0284c7;">${escapeHtml(prod.ten_sp)}</b> (${escapeHtml(prod.don_vi_tinh || "Cái")})</div>` +
        `<div style="margin-top:3px; font-size:12px; color:#475569;">🛣️ <b>Lộ trình:</b> ${escapeHtml(routeStr)}</div>`;
    });
  }
}

function renderMasterTable(data) {
  const container = el("master-list-container");
  if (!container) return;
  let html = `<table class="matrix-table" style="font-size: 13px;"><thead><tr><th>Mã SP</th><th>Tên Sản Phẩm</th><th>Lộ trình (Trạm)</th></tr></thead><tbody>`;
  data.forEach((p) => {
    let routingText =
      '<span style="color:#94a3b8;">Chỉ dùng cho BOM kho (Chưa gắn trạm SX)</span>';
    if (p.routing_map) {
      try {
        routingText = JSON.parse(p.routing_map)
          .map((step) => {
            const obj = state.globalStages.find(
              (st) => st.stage_id == step.stage_id,
            );
            return obj ? obj.stage_name : step.stage_id;
          })
          .join(" ➔ ");
      } catch (e) {
        routingText = "Lỗi";
      }
    }
    html += `<tr><td style="font-weight:bold; color:var(--navy);">${escapeHtml(p.ma_sp)}</td><td><b>${escapeHtml(p.ten_sp)}</b></td><td>${routingText}</td></tr>`;
  });
  container.innerHTML = html + `</tbody></table>`;
}

window.filterMasterData = function () {
  const val = (el("search-master")?.value || "").toLowerCase();
  renderMasterTable(
    state.productsCache.filter(
      (p) =>
        (p.ma_sp || "").toLowerCase().includes(val) ||
        (p.ten_sp || "").toLowerCase().includes(val),
    ),
  );
};

export function bindMasterDataForm(reload) {
  const form = el("form-create-product");
  if (!form) return;
  form.addEventListener("submit", async function (e) {
    e.preventDefault();
    const routingMap = [];
    document.querySelectorAll(".routing-stage-select").forEach((sel, idx) => {
      if (sel.value) routingMap.push({ order: idx + 1, stage_id: sel.value });
    });
    try {
      const payload = {
        ma_sp: el("md-product-code").value.trim().toUpperCase(),
        ten_sp: el("md-product-name").value.trim(),
        don_vi_tinh: "Cái",
      };
      if (routingMap.length > 0) payload.routing_map = routingMap;
      await api("/api/products", {
        method: "POST",
        body: JSON.stringify(payload),
      });
      showToast("Khởi tạo mã hàng thành công!");
      form.reset();
      el("routing-builder").innerHTML = "";
      routingStepCount = 0;
      await fetchProductsList();
      if (typeof reload === "function") await reload();
    } catch (err) {
      showToast(err.message, true);
    }
  });
}

export function bindPoCreateForm() {
  const form = el("form-create-po");
  if (!form) return;
  form.addEventListener("submit", async function (e) {
    e.preventDefault();
    if (!state.user) return;

    const prefix = el("po-prefix").value.trim().toUpperCase();
    const productCode = el("product-code").value.trim().toUpperCase();
    const poDate = el("po-date").value;
    const totalQty = parseInt(el("po-total").value);
    const splitQty = parseInt(el("po-split").value);

    const prod = (state.productsCache || []).find(
      (p) => String(p.ma_sp).toUpperCase() === productCode,
    );
    const spDisplay = prod ? `${productCode} - ${prod.ten_sp}` : productCode;

    if (splitQty > totalQty)
      return showToast(
        "Kích thước 1 LOT không được lớn hơn Tổng sản lượng!",
        true,
      );

    const lots = [];
    let remaining = totalQty;
    let counter = 1;
    while (remaining > 0) {
      const qtyForThisLot = remaining > splitQty ? splitQty : remaining;
      lots.push({
        po_number: `${prefix}-${String(counter).padStart(2, "0")}`,
        qty: qtyForThisLot,
      });
      remaining -= qtyForThisLot;
      counter++;
    }

    if (
      !confirm(
        `Mã hàng: ${spDisplay}\nHệ thống sẽ chẻ thành ${lots.length} LOT (${lots.map((l) => l.qty).join(", ")} pcs).\nBạn có chắc chắn?`,
      )
    )
      return;

    let successCount = 0;
    const errorMessages = [];

    for (const lot of lots) {
      try {
        await api("/api/sx/po/create", {
          method: "POST",
          body: JSON.stringify({
            po_number: lot.po_number,
            product_code: productCode,
            production_date: poDate,
            total_qty: lot.qty,
            user_id: state.user.ma_nv,
          }),
        });
        successCount++;
      } catch (err) {
        errorMessages.push(`Lỗi LOT ${lot.po_number}: ${err.message}`);
      }
    }

    if (errorMessages.length > 0) {
      alert(
        `Đã tạo được ${successCount}/${lots.length} LOT.\nDanh sách lỗi:\n` +
          errorMessages.join("\n"),
      );
    } else {
      showToast(
        `Đã cấp phát THÀNH CÔNG ${successCount}/${lots.length} LOT cho [${spDisplay}]!`,
      );
      form.reset();
      if (el("po-product-preview"))
        el("po-product-preview").style.display = "none";
      setSxDefaultDates();
    }

    window.fetchMatrixReport();
  });
}

// Mở Modal xem danh sách máy thuộc 1 Line (ở tab Line & Máy)
window.openLineMachinesModal = function (lineId) {
  ensureSxMachineModal();
  const line = state.globalLines.find((l) => l.line_id === lineId);
  const machines = state.globalMachines.filter((m) => m.line_id === lineId);
  const allAllocs = state.currentMatrixData?.allocations || [];

  el("sx-machine-modal-title").textContent =
    `🔧 Danh sách Máy thuộc ${line ? line.line_name : lineId} (${machines.length} máy)`;

  if (machines.length === 0) {
    el("sx-machine-modal-body").innerHTML =
      `<p style="text-align:center; color:#64748b; padding:20px;">Line này chưa khai báo máy nào.</p>`;
  } else {
    let html = `<table class="matrix-table" style="font-size:13px;"><thead><tr><th>STT</th><th>Mã Máy</th><th>Tên / Mô tả Máy</th><th>Trạng thái hiện tại</th></tr></thead><tbody>`;
    machines.forEach((m, idx) => {
      const running = allAllocs.filter(
        (a) => Number(a.machine_id) === Number(m.id) && a.status !== "DONE",
      );
      let statusHtml = `<span style="color:#64748b;">Đang trống</span>`;
      if (running.length > 0) {
        statusHtml = running
          .map(
            (r) =>
              `<div style="color:#0369a1; font-weight:bold;">⚙️ Đang chạy LOT ${escapeHtml(r.lot_number)} [${escapeHtml(r.ma_sp)}]: ${fmt(r.qty)} pcs (${calcDurationText(r.input_date, null)})</div>`,
          )
          .join("");
      }
      html += `<tr><td style="text-align:center;">${idx + 1}</td><td style="font-weight:bold; color:var(--navy);">${escapeHtml(m.machine_code)}</td><td>${escapeHtml(m.machine_name || "-")}</td><td>${statusHtml}</td></tr>`;
    });
    html += `</tbody></table>`;
    el("sx-machine-modal-body").innerHTML = html;
  }
  el("sx-machine-list-modal").classList.remove("hidden");
};

// Mở Modal xem chi tiết các máy đang chạy/đã chạy của 1 LOT tại 1 Công đoạn (trên bảng WIP)
window.openStageAllocModal = function (lotNumber, stageId) {
  ensureSxMachineModal();
  const allAllocs = (state.currentMatrixData?.allocations || []).filter(
    (a) =>
      a.lot_number === lotNumber &&
      String(a.stage_id).toUpperCase() === String(stageId).toUpperCase(),
  );
  el("sx-machine-modal-title").textContent =
    `🔧 Chi tiết Máy gia công — LOT: ${lotNumber} tại khâu ${stageName(stageId)}`;

  if (allAllocs.length === 0) {
    el("sx-machine-modal-body").innerHTML =
      `<p style="text-align:center; color:#64748b; padding:20px;">Chưa có máy nào được phân bổ ở công đoạn này.</p>`;
  } else {
    let html = `<table class="matrix-table" style="font-size:13px;"><thead><tr><th>Mã Máy</th><th>Mã SP</th><th>SL (pcs)</th><th>Bắt đầu</th><th>Kết thúc</th><th>Thời gian SX</th><th>Trạng thái</th></tr></thead><tbody>`;
    allAllocs.forEach((a) => {
      const isDone = a.status === "DONE";
      const dur = calcDurationText(a.input_date, isDone ? a.output_date : null);
      const badge = isDone
        ? `<span style="background:#dcfce7; color:#166534; padding:2px 6px; border-radius:4px; font-weight:bold;">✔ Đã hoàn thành</span>`
        : `<span style="background:#fef3c7; color:#b45309; padding:2px 6px; border-radius:4px; font-weight:bold;">⚙️ Đang sản xuất</span>`;
      html +=
        `<tr>` +
        `<td style="font-weight:bold; color:var(--navy);">${escapeHtml(a.machine_code || "-")}</td>` +
        `<td><b>${escapeHtml(a.ma_sp)}</b></td>` +
        `<td style="font-weight:bold; color:#0284c7;">${fmt(a.qty)}</td>` +
        `<td>${escapeHtml((a.input_date || "-").replace("T", " "))}</td>` +
        `<td>${escapeHtml((a.output_date || (isDone ? "-" : "Đang chạy...")).replace("T", " "))}</td>` +
        `<td style="font-weight:bold; color:#0f766e;">⏱️ ${dur}</td>` +
        `<td>${badge}</td>` +
        `</tr>`;
    });
    html += `</tbody></table>`;
    el("sx-machine-modal-body").innerHTML = html;
  }
  el("sx-machine-list-modal").classList.remove("hidden");
};

// ---------------- LINE / CÔNG ĐOẠN / MÁY ----------------
export async function renderLineMachineTable() {
  const container = el("linemay-list-container");
  if (!container) return;

  let html =
    '<div class="table-responsive"><table class="matrix-table" style="font-size:13px;"><thead><tr><th>Line</th><th>Công đoạn thuộc Line</th><th style="text-align:center;">Danh sách Máy thuộc Line</th></tr></thead><tbody>';
  if (state.globalLines.length === 0) {
    html +=
      '<tr><td colspan="3" style="text-align:center;">Chưa có Line nào.</td></tr>';
  } else {
    state.globalLines.forEach((line) => {
      const stages =
        state.globalStages
          .filter((s) => s.line_id === line.line_id)
          .map((s) => s.stage_name)
          .join(", ") || "-";
      const machineList = state.globalMachines.filter(
        (m) => m.line_id === line.line_id,
      );
      const machineBtnHtml =
        machineList.length > 0
          ? `<button type="button" class="btn btn-secondary" style="padding:5px 12px; font-size:12px; background:#0284c7; color:#fff;" onclick="openLineMachinesModal('${escapeHtml(line.line_id)}')">🔧 Xem danh sách (${machineList.length} máy)</button>`
          : `<span style="color:#94a3b8;">Chưa có máy</span>`;

      html += `<tr><td style="font-weight:bold; color:var(--navy);">${escapeHtml(line.line_name)} (${escapeHtml(line.line_id)})</td><td>${escapeHtml(stages)}</td><td style="text-align:center;">${machineBtnHtml}</td></tr>`;
    });
  }
  container.innerHTML = html + "</tbody></table></div>";

  const lineOpts =
    '<option value="">-- Chọn Line --</option>' +
    state.globalLines
      .map((l) => `<option value="${l.line_id}">${l.line_name}</option>`)
      .join("");
  if (el("new-stage-line")) el("new-stage-line").innerHTML = lineOpts;
  if (el("new-machine-line")) el("new-machine-line").innerHTML = lineOpts;
}

// ---------------- XỬ LÝ HÀNG NG (CHO PHÉP CHIA NHỎ SỐ LƯỢNG VỀ NHIỀU LINE HOẶC BÁO HỦY) ----------------
function currentSelectedNgData() {
  const val = el("ng-po-select")?.value;
  if (!val) return null;
  try {
    return JSON.parse(val);
  } catch (e) {
    return null;
  }
}

window.onNgLotSelectChange = function () {
  const ngInfo = currentSelectedNgData();
  const container = el("ng-split-container");
  const rowsWrap = el("ng-split-rows");
  if (!container || !rowsWrap) return;

  if (!ngInfo) {
    container.classList.add("hidden");
    rowsWrap.innerHTML = "";
    return;
  }

  container.classList.remove("hidden");
  el("ng-max-qty").textContent = ngInfo.ng_qty;
  rowsWrap.innerHTML = "";
  window.addNgSplitRow(ngInfo.ng_qty);
};

window.addNgSplitRow = function (defaultQty) {
  const rowsWrap = el("ng-split-rows");
  if (!rowsWrap) return;

  const stageOpts =
    '<option value="">-- Chọn công đoạn trả về (VD: NC1, Nhiệt luyện...) --</option>' +
    state.globalStages
      .map(
        (s) =>
          `<option value="${escapeHtml(s.stage_id)}">${escapeHtml(s.stage_name)} (${escapeHtml(s.stage_id)})</option>`,
      )
      .join("");

  const row = document.createElement("div");
  row.className = "ng-split-row";
  row.style.cssText =
    "display:flex; gap:8px; align-items:center; flex-wrap:wrap; background:#f8fafc; border:1px solid #cbd5e1; padding:10px; border-radius:8px; margin-bottom:8px;";
  row.innerHTML =
    `<input type="number" class="ng-row-qty" min="1" value="${defaultQty || 1}" placeholder="SL" style="width:85px; padding:8px; font-weight:bold;" oninput="updateNgSplitSum()" />` +
    `<select class="ng-row-action" style="flex:1; min-width:170px; padding:8px;" onchange="onNgRowActionChange(this)">` +
    `<option value="SCRAP">🗑️ Báo Hủy (SCRAP)</option>` +
    `<option value="REWORK">🔄 Làm lại (REWORK) - Chuyển về công đoạn</option>` +
    `</select>` +
    `<select class="ng-row-target hidden" style="flex:1.2; min-width:190px; padding:8px;">${stageOpts}</select>` +
    `<button type="button" class="btn btn-danger" style="padding:6px 10px;" onclick="this.parentElement.remove(); updateNgSplitSum();">✕</button>`;

  rowsWrap.appendChild(row);
  window.updateNgSplitSum();
};

window.onNgRowActionChange = function (selectElem) {
  const row = selectElem.closest(".ng-split-row");
  const targetSelect = row.querySelector(".ng-row-target");
  if (selectElem.value === "REWORK") {
    targetSelect.classList.remove("hidden");
  } else {
    targetSelect.classList.add("hidden");
  }
};

window.updateNgSplitSum = function () {
  const ngInfo = currentSelectedNgData();
  const max = ngInfo ? Number(ngInfo.ng_qty) : 0;
  let sum = 0;
  document.querySelectorAll("#ng-split-rows .ng-row-qty").forEach((inp) => {
    sum += Number(inp.value) || 0;
  });
  if (el("ng-selected-total")) el("ng-selected-total").textContent = sum;
  if (el("ng-remain-qty")) el("ng-remain-qty").textContent = max - sum;
};

    const rowElems = Array.from(

    if (rowElems.length === 0)
      return showToast("Hãy thêm ít nhất 1 dòng phương án xử lý.", true);

    const actions = [];
    let totalSelected = 0;

    for (const r of rowElems) {
      const qty = parseInt(r.querySelector(".ng-row-qty").value) || 0;
      const action = r.querySelector(".ng-row-action").value;
      const returnStage = r.querySelector(".ng-row-target").value;
      if (qty <= 0)
        return showToast("Số lượng ở mỗi dòng phải lớn hơn 0.", true);
      if (action === "REWORK" && !returnStage) {
        return showToast(
          "Vui lòng chọn công đoạn trả về (VD: NC1, Nhiệt luyện...) cho dòng Làm lại!",
          true,
        );
      }
      totalSelected += qty;
      actions.push({
        qty,
        action,
        return_stage: action === "REWORK" ? returnStage : "",
      });
    }

    if (totalSelected > ngInfo.ng_qty) {
      return showToast(
        `Tổng số lượng chia xử lý (${totalSelected}) vượt quá số lượng NG hiện có (${ngInfo.ng_qty})!`,
        true,
      );
    }

    try {
      for (const item of actions) {
        await api("/api/sx/ng/resolve", {
          method: "POST",
          body: JSON.stringify({
            lot_number: ngInfo.lot_number,
            from_stage: ngInfo.from_stage,
            qty: item.qty,
            action: item.action,
            return_stage: item.return_stage,
            user_id: state.user.ma_nv,
          }),
        });
      }
      showToast(
        `Đã xử lý thành công ${totalSelected} pcs hàng NG của LOT ${ngInfo.lot_number}!`,
      );
      form.reset();
      if (el("ng-split-container"))
        el("ng-split-container").classList.add("hidden");
      await loadStagesLinesMachines();
      window.fetchMatrixReport();
      if (typeof reload === "function") await reload();
    } catch (err) {
      showToast(err.message, true);
    }
  });
}
